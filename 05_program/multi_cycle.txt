# Khởi tạo hằng và địa chỉ
li      x1, 0x3f800000        # 1.0
li      x2, 0x40000000        # 2.0
li      x31, 0x7000           # địa chỉ xuất kết quả

# Nạp vào các thanh ghi FP
fmv.s.x f0, x1                # f0 = 1.0
fmv.s.x f1, x2                # f1 = 2.0

# Chia multi-cycle
fdiv.s    f4, f1, f0          # 2/1 = 2.0
fsw       f4, 0(x31)

done:
jal       x0, done
